from risk_engine import RiskEngine
from audit_trail import AuditLogger

def on_telemetry_packet(event):
    # Stateful transition check
    previous_score = state_store.get(
        event['asset_id']
    )
    assessment = RiskEngine().evaluate_asset(event)

    if assessment['risk_score'] > previous_score + 15.0:
        dispatch_early_warning(assessment)

    AuditLogger().log_assessment(event, assessment)
